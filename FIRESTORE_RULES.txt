Firebase Console > Firestore Database > Rules में नीचे के rules लगाएँ और Publish करें.
ये rules केवल customer-side appointment और order create करने देते हैं. Public read, update, delete बंद हैं.
Admin dashboard अभी सुरक्षित रूप से cloud-connected नहीं है; उसे Firebase Authentication के साथ अलग से secure करना होगा.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function validPhone(p) {
      return p is string && p.matches('^[0-9]{10}$');
    }
    match /appointments/{id} {
      allow create: if
        request.resource.data.keys().hasAll([
          'customerName','phone','serviceName','price','date','time','status','createdAt'
        ]) &&
        request.resource.data.customerName is string &&
        request.resource.data.customerName.size() > 0 &&
        validPhone(request.resource.data.phone) &&
        request.resource.data.serviceName is string &&
        request.resource.data.price is number &&
        request.resource.data.date is string &&
        request.resource.data.time is string &&
        request.resource.data.status == 'pending';
      allow read, update, delete: if false;
    }
    match /orders/{id} {
      allow create: if
        request.resource.data.keys().hasAll([
          'orderId','customerName','phone','address','paymentMethod',
          'deliveryMethod','total','items','status','createdAt'
        ]) &&
        request.resource.data.customerName is string &&
        request.resource.data.customerName.size() > 0 &&
        validPhone(request.resource.data.phone) &&
        request.resource.data.address is string &&
        request.resource.data.address.size() > 5 &&
        request.resource.data.paymentMethod in ['UPI','Cash on Delivery'] &&
        request.resource.data.deliveryMethod in ['Courier','Self Shipping'] &&
        request.resource.data.total is number &&
        request.resource.data.total > 0 &&
        request.resource.data.items is list &&
        request.resource.data.status == 'pending';
      allow read, update, delete: if false;
    }
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
